#        colcon build --packages-select tiago_group_approach

#    Kept out on purpose: it pins numpy==1.25.0, which directly conflicts

#    environment and requirements-locateanything.txt instead.

#    any script in this repo (verified). Left out to keep the install lean.

# --- Machine learning -------------------------------------------------------

scikit-learn==1.7.2
joblib==1.5.3


# --- scipy: PIN IT, DO NOT RELY ON THE SYSTEM PACKAGE ------------------------
# scikit-learn imports scipy.sparse, so scipy is a transitive dependency and
# pip will happily accept whatever is already installed. On the ROS 2 Humble
# devcontainer that is the apt package scipy 1.8.0 in
# /usr/lib/python3/dist-packages, which was compiled against NumPy 1.x. With
# the numpy 2.2.6 pinned above, importing sklearn then dies with:
#
#     A module that was compiled using NumPy 1.x cannot be run in NumPy 2.2.6
#     AttributeError: _ARRAY_API not found
#     ImportError: numpy.core.multiarray failed to import
#
# pip reports "Requirement already satisfied: scipy>=1.8.0" and moves on, so
# nothing looks wrong until sklearn is first imported. Pinning a NumPy-2 build
# forces a user-site install that shadows the stale system one.
#
# This exact combination is verified working on Python 3.10.12:
#     numpy 2.2.6 / scipy 1.15.3 / scikit-learn 1.7.2 / pandas 2.3.3
scipy==1.15.3


# --- matplotlib: only for the Chapter 5 figures ------------------------------
# Used by scripts/plot_results.py to draw the approach-accuracy figures. It is
# NOT needed to run the simulation, train the models or produce the numeric
# results - only to render them.
#
# It does not depend on scipy, so installing it cannot re-trigger the numpy /
# scipy ABI clash documented above.
matplotlib==3.10.9

# --- Person detection (YOLOv8n) ---------------------------------------------
# This is the project's adopted perception model (99.7% recall on the
# sessions 1/3 ground truth vs. the proposal's 80% target).
#   extract_person_detections.py, test_person_detector.py,
#   validate_detector_recall.py
#
# WARNING - INSTALL SIZE. ultralytics depends on PyTorch, and on Linux pip
# resolves that to the CUDA build, which drags in the whole NVIDIA stack
# (cuBLAS, cuDNN, NCCL, cuFFT, ...). Verified by dry-run: that is roughly
# 3-4 GB and the slowest part of the install by a wide margin.
#
# You very likely do NOT need it. Person detection has already been run
# across all 24 sessions and cached to detected_people.csv and
# detected_people_individual.csv, so every downstream step (clustering,
# O-space, training, evaluation) works without ultralytics installed at all.
# It is only required to RE-RUN perception from scratch.
#
# Two ways to avoid the multi-GB download:
#
#   (a) Skip it entirely - comment out the line below. Everything except
#       extract_person_detections.py / test_person_detector.py still runs.
#
#   (b) Install the CPU-only PyTorch first (~200 MB instead of ~4 GB), then
#       this file. YOLOv8n is small and runs fine on CPU:
#           pip install torch torchvision --index-url https://download.pytorch.org/whl/cpu
#           pip install -r requirements.txt
ultralytics>=8.3.0,<9.0

rosbags==0.11.3
