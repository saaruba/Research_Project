# --- Core -------------------------------------------------------------------
# Image conversion in group_perception_node, depth back-projection, geometry.
numpy==2.2.6


# --- Person detection -------------------------------------------------------
# YOLOv8n, the detector used for every experiment run (99.7% recall on the
# sessions 1/3 ground truth against the proposal's 80% target).
#
# INSTALL SIZE WARNING: on Linux, pip resolves ultralytics' PyTorch dependency
# to the CUDA build and pulls in the whole NVIDIA stack - roughly 3-4 GB. If
# your lab PC has no GPU, or you simply do not want the wait, install the
# CPU-only PyTorch FIRST and this file second:
#
#     pip install torch torchvision --index-url https://download.pytorch.org/whl/cpu
#     pip install -r requirements-simulation.txt
#
# YOLOv8n is tiny and runs comfortably on CPU at the ~2 Hz this pipeline uses.
ultralytics>=8.3.0,<9.0


# --- Detection overlay ------------------------------------------------------
# Draws the boxes on /perception/image_annotated.
# Headless build: the container has no display of its own. If you want
# cv2.imshow() windows on a desktop machine, swap for opencv-python -
# install ONE of them, never both.
opencv-python-headless==4.13.0.92


# --- Loading the trained BC model -------------------------------------------
# bc_policy_node loads approach_pose_random_forest_tuned.joblib. scikit-learn
# must be present to UNPICKLE it even though no training happens at runtime,
# and the version must match the one that wrote the file or you will get
# InconsistentVersionWarning and possibly wrong predictions.
scikit-learn==1.7.2
joblib==1.5.3

# scipy is pulled in by scikit-learn. It MUST be pinned: the devcontainer's
# apt scipy 1.8.0 was built against NumPy 1.x and crashes on import with
# "_ARRAY_API not found" under the numpy 2.2.6 above. pip will not replace it
# on its own, because "scipy>=1.8.0" is technically satisfied.
scipy==1.15.3
